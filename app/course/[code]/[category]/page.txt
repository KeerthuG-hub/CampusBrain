'use client';

import { useState, useEffect } from 'react';
import { useParams, useRouter } from 'next/navigation';
import { ArrowLeft, Download, Eye, Filter, X, Search, Calendar, User, Tag } from 'lucide-react';
import Link from 'link';
import { supabase } from '@/lib/supabase/client';
import { Course, ResourceWithDetails, CATEGORY_LABELS } from '@/lib/types';

export default function ResourcesPage() {
  const params = useParams();
  const router = useRouter();
  const courseCode = params.code as string;
  const category = params.category as string;

  const [course, setCourse] = useState<Course | null>(null);
  const [resources, setResources] = useState<ResourceWithDetails[]>([]);
  const [filteredResources, setFilteredResources] = useState<ResourceWithDetails[]>([]);
  const [loading, setLoading] = useState(true);
  
  // Filter states
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedTags, setSelectedTags] = useState<string[]>([]);
  const [selectedCredits, setSelectedCredits] = useState<number[]>([]);
  const [selectedCourseModes, setSelectedCourseModes] = useState<string[]>([]);
  const [sortBy, setSortBy] = useState<'recent' | 'popular' | 'name'>('recent');
  const [showFilters, setShowFilters] = useState(true);

  // Available tags (fetch from DB or hardcode)
  const [availableTags, setAvailableTags] = useState<string[]>([]);

  useEffect(() => {
    fetchData();
  }, [courseCode, category]);

  useEffect(() => {
    applyFilters();
  }, [resources, searchTerm, selectedTags, sortBy]);

  async function fetchData() {
    setLoading(true);

    // Fetch course
    const { data: courseData } = await supabase
      .from('courses')
      .select('*')
      .eq('code', courseCode)
      .single();

    if (courseData) {
      setCourse(courseData);

      // Determine resource filter based on category
      let resourceQuery = supabase
        .from('resources')
        .select(`
          *,
          course:courses(*),
          uploader:profiles(full_name, email),
          storage_mapping:storage_mappings(external_link, provider)
        `)
        .eq('course_id', courseData.id)
        .eq('is_approved', true);

      // Filter by category
      if (category === 'course-plan') {
        resourceQuery = resourceQuery.like('title', '[Course Plan]%');
      } else if (category === 'notes-ppts') {
        resourceQuery = resourceQuery.in('resource_type', ['notes', 'ppt']);
      } else if (category === 'books') {
        resourceQuery = resourceQuery.eq('resource_type', 'book');
      } else if (category === 'question-papers') {
        resourceQuery = resourceQuery.eq('resource_type', 'form');
      }

      const { data: resourcesData } = await resourceQuery.order('created_at', { ascending: false });

      setResources(resourcesData || []);
      
      // Extract unique tags (example - you'd fetch from tags table)
      const tags = new Set<string>();
      resourcesData?.forEach(r => {
        // Mock tags - replace with actual tag fetching
        if (r.file_type?.includes('pdf')) tags.add('PDF');
        if (r.file_type?.includes('ppt')) tags.add('PPT');
        if (r.file_type?.includes('doc')) tags.add('DOC');
      });
      setAvailableTags(Array.from(tags));
    }

    setLoading(false);
  }

  function applyFilters() {
    let filtered = [...resources];

    // Search filter
    if (searchTerm) {
      filtered = filtered.filter(r =>
        r.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
        r.description?.toLowerCase().includes(searchTerm.toLowerCase())
      );
    }

    // Tag filter
    if (selectedTags.length > 0) {
      filtered = filtered.filter(r =>
        selectedTags.some(tag => r.file_type?.toUpperCase().includes(tag))
      );
    }

    // Sort
    if (sortBy === 'recent') {
      filtered.sort((a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime());
    } else if (sortBy === 'popular') {
      filtered.sort((a, b) => (b.download_count + b.view_count) - (a.download_count + a.view_count));
    } else if (sortBy === 'name') {
      filtered.sort((a, b) => a.title.localeCompare(b.title));
    }

    setFilteredResources(filtered);
  }

  const toggleTag = (tag: string) => {
    setSelectedTags(prev =>
      prev.includes(tag) ? prev.filter(t => t !== tag) : [...prev, tag]
    );
  };

  const handleDownload = async (resourceId: string, externalLink: string) => {
    await supabase.rpc('increment_download_count', { resource_id: resourceId });
    window.open(externalLink, '_blank');
  };

  const handleView = async (resourceId: string, externalLink: string) => {
    await supabase.rpc('increment_view_count', { resource_id: resourceId });
    const previewUrl = externalLink.replace('/view', '/preview');
    window.open(previewUrl, '_blank');
  };

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div>
      </div>
    );
  }

  const categoryLabel = CATEGORY_LABELS[category as keyof typeof CATEGORY_LABELS] || category;

  return (
    <div className="min-h-screen bg-gray-50">
      {/* Header */}
      <header className="bg-white shadow-sm border-b sticky top-0 z-20">
        <div className="max-w-7xl mx-auto px-4 py-4">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-4">
              <Link
                href={`/course/${courseCode}`}
                className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
              >
                <ArrowLeft className="w-5 h-5" />
              </Link>
              <div>
                <div className="flex items-center gap-2 mb-1">
                  <span className="text-sm text-gray-500">{course?.code}</span>
                  <span className="text-gray-300">•</span>
                  <span className="text-sm font-semibold text-blue-600">{categoryLabel}</span>
                </div>
                <h1 className="text-lg font-bold text-gray-900">
                  {course?.name}
                </h1>
              </div>
            </div>

            {/* Mobile Filter Toggle */}
            <button
              onClick={() => setShowFilters(!showFilters)}
              className="lg:hidden p-2 hover:bg-gray-100 rounded-lg"
            >
              <Filter className="w-5 h-5" />
            </button>
          </div>
        </div>
      </header>

      {/* Main Layout */}
      <div className="max-w-7xl mx-auto px-4 py-6">
        <div className="flex gap-6">
          {/* FILTERS SIDEBAR (Amazon style!) */}
          {showFilters && (
            <aside className="w-64 flex-shrink-0 space-y-6">
              {/* Search */}
              <div className="bg-white rounded-lg shadow-sm border p-4">
                <h3 className="font-semibold text-gray-900 mb-3 flex items-center gap-2">
                  <Search className="w-4 h-4" />
                  Search
                </h3>
                <input
                  type="text"
                  placeholder="Search resources..."
                  value={searchTerm}
                  onChange={(e) => setSearchTerm(e.target.value)}
                  className="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
                />
              </div>

              {/* Sort */}
              <div className="bg-white rounded-lg shadow-sm border p-4">
                <h3 className="font-semibold text-gray-900 mb-3">Sort By</h3>
                <div className="space-y-2">
                  {[
                    { value: 'recent', label: 'Most Recent' },
                    { value: 'popular', label: 'Most Popular' },
                    { value: 'name', label: 'Name (A-Z)' }
                  ].map(option => (
                    <label key={option.value} className="flex items-center gap-2 cursor-pointer">
                      <input
                        type="radio"
                        name="sort"
                        value={option.value}
                        checked={sortBy === option.value}
                        onChange={(e) => setSortBy(e.target.value as any)}
                        className="text-blue-600 focus:ring-blue-500"
                      />
                      <span className="text-sm text-gray-700">{option.label}</span>
                    </label>
                  ))}
                </div>
              </div>

              {/* File Type Filter */}
              {availableTags.length > 0 && (
                <div className="bg-white rounded-lg shadow-sm border p-4">
                  <h3 className="font-semibold text-gray-900 mb-3 flex items-center gap-2">
                    <Tag className="w-4 h-4" />
                    File Type
                  </h3>
                  <div className="space-y-2">
                    {availableTags.map(tag => (
                      <label key={tag} className="flex items-center gap-2 cursor-pointer">
                        <input
                          type="checkbox"
                          checked={selectedTags.includes(tag)}
                          onChange={() => toggleTag(tag)}
                          className="rounded text-blue-600 focus:ring-blue-500"
                        />
                        <span className="text-sm text-gray-700">{tag}</span>
                      </label>
                    ))}
                  </div>
                </div>
              )}

              {/* Active Filters */}
              {(searchTerm || selectedTags.length > 0) && (
                <div className="bg-blue-50 rounded-lg border border-blue-200 p-4">
                  <div className="flex items-center justify-between mb-2">
                    <h3 className="font-semibold text-blue-900 text-sm">Active Filters</h3>
                    <button
                      onClick={() => {
                        setSearchTerm('');
                        setSelectedTags([]);
                      }}
                      className="text-xs text-blue-600 hover:text-blue-800"
                    >
                      Clear All
                    </button>
                  </div>
                  <div className="space-y-1">
                    {searchTerm && (
                      <div className="flex items-center gap-2 text-sm text-blue-900">
                        <Search className="w-3 h-3" />
                        <span>"{searchTerm}"</span>
                      </div>
                    )}
                    {selectedTags.map(tag => (
                      <div key={tag} className="flex items-center gap-2 text-sm text-blue-900">
                        <Tag className="w-3 h-3" />
                        <span>{tag}</span>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </aside>
          )}

          {/* RESOURCES GRID */}
          <main className="flex-1">
            {/* Results Header */}
            <div className="mb-6 flex items-center justify-between">
              <p className="text-sm text-gray-600">
                Showing <span className="font-semibold">{filteredResources.length}</span> of{' '}
                <span className="font-semibold">{resources.length}</span> resources
              </p>
            </div>

            {/* Resources Grid */}
            {filteredResources.length === 0 ? (
              <div className="text-center py-16 bg-white rounded-lg">
                <Search className="w-16 h-16 text-gray-400 mx-auto mb-4" />
                <h3 className="text-lg font-semibold text-gray-900 mb-2">
                  No resources found
                </h3>
                <p className="text-gray-600 mb-4">
                  Try adjusting your filters or search term
                </p>
                <button
                  onClick={() => {
                    setSearchTerm('');
                    setSelectedTags([]);
                  }}
                  className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700"
                >
                  Clear Filters
                </button>
              </div>
            ) : (
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                {filteredResources.map((resource) => (
                  <div
                    key={resource.id}
                    className="bg-white rounded-lg shadow-sm border hover:shadow-md transition-all p-5"
                  >
                    {/* File Type Badge */}
                    <div className="inline-block px-2 py-1 bg-blue-100 text-blue-700 text-xs font-semibold rounded mb-3">
                      {resource.file_type?.split('/')[1]?.toUpperCase() || 'FILE'}
                    </div>

                    {/* Title */}
                    <h3 className="font-semibold text-gray-900 mb-2 line-clamp-2 min-h-[3rem]">
                      {resource.title.replace('[Course Plan] ', '')}
                    </h3>

                    {/* Description */}
                    {resource.description && (
                      <p className="text-sm text-gray-600 mb-3 line-clamp-2">
                        {resource.description}
                      </p>
                    )}

                    {/* Stats */}
                    <div className="flex items-center gap-4 text-sm text-gray-500 mb-4">
                      <span className="flex items-center gap-1">
                        <Eye className="w-4 h-4" />
                        {resource.view_count}
                      </span>
                      <span className="flex items-center gap-1">
                        <Download className="w-4 h-4" />
                        {resource.download_count}
                      </span>
                    </div>

                    {/* Actions */}
                    <div className="flex gap-2">
                      <button
                        onClick={() => handleView(resource.id, resource.storage_mapping?.external_link || resource.file_url)}
                        className="flex-1 px-3 py-2 border border-gray-300 rounded-lg hover:bg-gray-50 transition-colors text-sm font-medium"
                      >
                        View
                      </button>
                      <button
                        onClick={() => handleDownload(resource.id, resource.storage_mapping?.external_link || resource.file_url)}
                        className="flex-1 px-3 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors text-sm font-medium"
                      >
                        Download
                      </button>
                    </div>

                    {/* Footer */}
                    <div className="mt-4 pt-4 border-t flex items-center justify-between text-xs text-gray-500">
                      <span className="flex items-center gap-1">
                        <User className="w-3 h-3" />
                        {resource.uploader?.full_name || 'Anonymous'}
                      </span>
                      <span className="flex items-center gap-1">
                        <Calendar className="w-3 h-3" />
                        {new Date(resource.created_at).toLocaleDateString()}
                      </span>
                    </div>
                  </div>
                ))}
              </div>
            )}
          </main>
        </div>
      </div>
    </div>
  );
}
